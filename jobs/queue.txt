# 训练实验队列模板
# 规则：
# 1. 每行一条完整命令
# 2. 以 # 开头的行为注释
# 3. 运行前请先替换每个数据集对应的均值、标准差和路径占位符
# 4. 除了模型、数据集名、通道数、mean/std、Train/Val 路径外，其余参数全部走 configs/train.yaml 默认值
#
# 数据集占位符约定：
# - wind / mwp / swh: 单通道，mean/std 各 1 个值
# - mwd: 双通道，mean/std 各 2 个值
#
# wind:
#   <WIND_MEAN> <WIND_STD>
#   <WIND_TRAIN_LR_ROOT> <WIND_TRAIN_HR_ROOT> <WIND_VAL_LR_ROOT> <WIND_VAL_HR_ROOT>
# mwp:
#   <MWP_MEAN> <MWP_STD>
#   <MWP_TRAIN_LR_ROOT> <MWP_TRAIN_HR_ROOT> <MWP_VAL_LR_ROOT> <MWP_VAL_HR_ROOT>
# swh:
#   <SWH_MEAN> <SWH_STD>
#   <SWH_TRAIN_LR_ROOT> <SWH_TRAIN_HR_ROOT> <SWH_VAL_LR_ROOT> <SWH_VAL_HR_ROOT>
# mwd:
#   <MWD_MEAN_0> <MWD_MEAN_1> <MWD_STD_0> <MWD_STD_1>
#   <MWD_TRAIN_LR_ROOT> <MWD_TRAIN_HR_ROOT> <MWD_VAL_LR_ROOT> <MWD_VAL_HR_ROOT>

# ====================
# EDSR
# ====================
uv run python scripts/train.py models=edsr dataset.name=wind dataset.channels=1 dataset.normalize.mean=[<WIND_MEAN>] dataset.normalize.std=[<WIND_STD>] dataset.train_lr_root="<WIND_TRAIN_LR_ROOT>" dataset.train_hr_root="<WIND_TRAIN_HR_ROOT>" dataset.val_lr_root="<WIND_VAL_LR_ROOT>" dataset.val_hr_root="<WIND_VAL_HR_ROOT>"
